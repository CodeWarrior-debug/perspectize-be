         ```typescript
         import { createMutation, useQueryClient } from '@tanstack/svelte-query';
         import { toast } from 'svelte-sonner';
         import { graphqlClient } from '../client';
         import { SET_PRIMARY_CATEGORY, type SetPrimaryCategoryResponse } from '../categories';
         import { queryKeys } from '../keys';

         export interface SetPrimaryCategoryInput {
             contentId: number;
             qid: string;
             label: string;
             description?: string;
             entityType?: string;
         }

         export function useSetPrimaryCategory() {
             const queryClient = useQueryClient();
             return createMutation(() => ({
                 mutationFn: async (input: SetPrimaryCategoryInput) => {
                     return graphqlClient.request<SetPrimaryCategoryResponse>(SET_PRIMARY_CATEGORY, {
                         input: {
                             contentId: input.contentId,
                             qid: input.qid,
                             label: input.label,
                             description: input.description ?? '',
                             entityType: input.entityType ?? '',
                         },
                     });
                 },
                 onSuccess: (data) => {
                     const category = data?.setPrimaryCategory?.primaryCategory;
                     toast.success(`Category set: ${category?.label ?? 'unknown'}`);
                     // Invalidate content list queries for table refresh
                     queryClient.invalidateQueries({ queryKey: queryKeys.content.lists() });
                 },
                 onError: (err: Error) => {
                     console.error('[SetPrimaryCategory] mutation failed:', err);
                     toast.error('Failed to set category. Please try again.');
                 },
             }));
         }
         ```

  <action>
    Create `frontend/src/lib/components/CategoryTypeahead.svelte` — a standalone typeahead search component for Wikidata entities that will be used inside a popover in Plan 04.

    **Props (Svelte 5 runes):**
    ```typescript
    let {
        contentId,
        currentCategory = null,
        onSelect,
        onClose,
    }: {
        contentId: number;
        currentCategory: { label: string; wikidataQid: string } | null;
        onSelect: (result: WikidataSearchResult) => void;
        onClose: () => void;
    } = $props();
    ```

    **State:**
    ```typescript
    let searchTerm = $state('');
    let debouncedTerm = $state('');
    ```

    **Debounce (300ms):**
    Use `$effect` with setTimeout/clearTimeout to debounce searchTerm into debouncedTerm:
    ```typescript
    $effect(() => {
        const timer = setTimeout(() => { debouncedTerm = searchTerm; }, 300);
        return () => clearTimeout(timer);
    });
    ```

    **TanStack Query for search results:**
    ```typescript
    const searchQuery = createQuery(() => ({
        queryKey: queryKeys.categories.search(debouncedTerm),
        queryFn: () => graphqlClient.request<WikidataSearchResponse>(WIKIDATA_SEARCH, {
            query: debouncedTerm,
            language: 'en',
            limit: 10,
        }),
        enabled: debouncedTerm.length >= 2,  // Minimum 2 chars (pitfall #6)
        placeholderData: keepPreviousData,     // Smooth typeahead UX
        staleTime: 5 * 60 * 1000,             // Cache results 5 min
    }));
    ```

    **Template structure:**
    ```svelte
    <div class="category-typeahead w-64 p-2">
        <!-- Search input -->
        <input
            type="text"
            bind:value={searchTerm}
            placeholder="Search Wikidata..."
            class="w-full rounded border border-input px-3 py-1.5 text-sm outline-none focus:ring-1 focus:ring-ring"
            autofocus
        />

        <!-- Results list -->
        {#if searchQuery.isLoading && debouncedTerm.length >= 2}
            <div class="p-2 text-sm text-muted-foreground">Searching...</div>
        {:else if searchQuery.data?.wikidataSearch?.length}
            <ul class="mt-1 max-h-48 overflow-y-auto">
                {#each searchQuery.data.wikidataSearch as result}
                    <li>
                        <button
                            type="button"
                            class="w-full px-2 py-1.5 text-left text-sm hover:bg-muted rounded cursor-pointer"
                            onclick={() => onSelect(result)}
                        >
                            <div class="font-medium">{result.label}</div>
                            {#if result.description}
                                <div class="text-xs text-muted-foreground truncate">{result.description}</div>
                            {/if}
                        </button>
                    </li>
                {/each}
            </ul>
        {:else if debouncedTerm.length >= 2 && !searchQuery.isLoading}
            <div class="p-2 text-sm text-muted-foreground">No results found</div>
        {:else if debouncedTerm.length > 0 && debouncedTerm.length < 2}
            <div class="p-2 text-sm text-muted-foreground">Type at least 2 characters</div>
        {/if}

        <!-- Current category display -->
        {#if currentCategory}
            <div class="mt-2 border-t pt-2 text-xs text-muted-foreground">
                Current: {currentCategory.label} ({currentCategory.wikidataQid})
            </div>
        {/if}
    </div>
    ```

    **Key design decisions:**
    - Component is a "dumb" search UI — it does NOT call the mutation itself
    - Parent (ActivityTable popover) handles mutation call via onSelect callback
    - This keeps the component reusable and testable
    - The 300ms debounce and 2-char minimum prevent excessive API calls (pitfall #6)
    - `keepPreviousData` from TanStack Query prevents flicker during typing
    - `staleTime: 5 minutes` caches repeated searches
