    ALTER TABLE perspectives ADD COLUMN related_perspective_ids INTEGER[] DEFAULT '{}';
    ALTER TABLE perspectives ADD COLUMN custom_fields JSONB DEFAULT '{}';
    ALTER TABLE perspectives ADD COLUMN review TEXT;

    -- GIN index for reverse lookups on related_perspective_ids
    CREATE INDEX idx_perspectives_related_ids ON perspectives USING GIN (related_perspective_ids);

    Define GraphQL operations:
    - `CREATE_PERSPECTIVE` mutation — accepts CreatePerspectiveInput, returns full PerspectiveItem fields
    - `UPDATE_PERSPECTIVE` mutation — accepts UpdatePerspectiveInput, returns full PerspectiveItem fields
    - `LIST_PERSPECTIVES_BY_USER` query — uses `perspectives(filter: { userID: $userID })` with no pagination (user typically has few perspectives), returns items array

    Response interfaces:
    - `CreatePerspectiveResponse { createPerspective: PerspectiveItem }`
    - `UpdatePerspectiveResponse { updatePerspective: PerspectiveItem }`
    - `ListPerspectivesByUserResponse { perspectives: { items: PerspectiveItem[] } }`
