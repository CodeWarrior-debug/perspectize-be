    Rating conversion utilities:
    ```typescript
    /** Convert storage value (0-10000) to display value (0.00-10.00) */
    export function ratingToDisplay(value: number): string {
      return (value / 1000).toFixed(2);
    }

    /** Convert display value (0.00-10.00) to storage value (0-10000) */
    export function displayToRating(display: number): number {
      return Math.round(display * 1000);
    }

    /** Validate a storage rating is within range */
    export function isValidRating(value: number): boolean {
      return Number.isInteger(value) && value >= 0 && value <= 10000;
    }

    /** Step amount for rating inputs — 0.25 display = 250 storage */
    export const RATING_STEP = 0.25;
    export const RATING_MIN = 0;
    export const RATING_MAX = 10;
    ```

    Add to `frontend/src/lib/utils/formatting.ts`:

    This is an AG Grid cell renderer function (returns HTMLElement, NOT a Svelte component):

    ```typescript
    export function perspectiveCellRenderer(params: any): HTMLElement {
      const container = document.createElement('div');
      container.className = 'h-full w-full flex items-center justify-center';

      // Check if user has perspective on this content (via params.context)
      const hasPerspective = params.context?.perspectivesByContentId?.has(params.data?.id);

      if (hasPerspective) {
        // Silhouette-with-glasses SVG icon (inline SVG, ~20x20)
        container.innerHTML = `<svg ...silhouette-with-glasses...>`;
        container.title = 'Edit your perspective';
      } else {
        // Plus icon
        const icon = document.createElement('span');
        icon.textContent = '+';
        icon.className = 'text-lg font-semibold text-primary hover:text-primary/80';
        container.appendChild(icon);
        container.title = 'Add a perspective';
      }

      return container;
    }
    ```
